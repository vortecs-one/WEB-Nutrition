  const sectionTitle = "text-xs font-semibold uppercase tracking-wide text-sidebar-foreground/70";

  return (
    <Modal isOpen={isOpen} onClose={onClose} title={t.title}>
      <div className="flex flex-col gap-4">
        <div className="flex flex-col gap-2">
          <div className="grid grid-cols-2 gap-3 sm:gap-4">
            <section
              aria-labelledby={`${ids}-current`}
              className="flex min-w-0 flex-col justify-between gap-3 rounded-2xl bg-sidebar-accent p-3 sm:p-4"
            >
              <h3 id={`${ids}-current`} className={sectionTitle}>
                {t.currentGlucose}
              </h3>
              {currentMgdl !== null ? (
                <div className="flex flex-col gap-2">
                  <p className="flex items-baseline gap-1.5 text-3xl font-semibold tabular-nums sm:text-4xl">
                    {formatGlucose(currentMgdl, unit)}
                    <span className="text-sm font-normal text-sidebar-foreground/70 sm:text-base">
                      {glucoseUnit}
                    </span>
                  </p>
                  {isStale && (
                    <p className="flex items-start gap-1.5 text-xs text-amber-500 sm:text-sm">
                      <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0" aria-hidden="true" />
                      {t.staleWarning}
                    </p>
                  )}
                </div>
              ) : (
                <div className="flex flex-col gap-2">
                  <p className="text-xs text-sidebar-foreground/70 sm:text-sm">{t.noReading}</p>
                  <NumberField
                    id={`${ids}-manual`}
                    label={t.manualGlucose}
                    suffix={glucoseUnit}
                    value={manualGlucose}
                    onChange={setManualGlucose}
                    step={glucoseStep}
                  />
                </div>
              )}
            </section>

            <div className="flex min-w-0 flex-col justify-between gap-3 rounded-2xl bg-sidebar-accent p-3 focus-within:ring-2 focus-within:ring-primary sm:p-4">
              <label htmlFor={`${ids}-carbs`} className={sectionTitle}>
                {t.carbsToEat}
              </label>
              <div className="flex items-baseline gap-1.5">
                <input
                  id={`${ids}-carbs`}
                  type="number"
                  inputMode="decimal"
                  min="0"
                  step="1"
                  value={carbs}
                  onChange={(e) => setCarbs(e.target.value)}
                  onFocus={(e) => e.target.select()}
                  aria-describedby={`${ids}-carbs-hint`}
                  className="w-full min-w-0 bg-transparent text-3xl font-semibold tabular-nums outline-none sm:text-4xl"
                />
                <span className="shrink-0 text-sm text-sidebar-foreground/70 sm:text-base">{t.grams}</span>
              </div>
            </div>
          </div>
          <p id={`${ids}-carbs-hint`} className="px-1 text-xs text-sidebar-foreground/60">
            {t.carbsHint}
          </p>
        </div>

        <div className="grid grid-cols-2 items-stretch gap-3 sm:gap-4">
          <section
            aria-labelledby={`${ids}-result`}
            aria-live="polite"
            className="flex min-w-0 flex-col gap-3"
          >
            <h3 id={`${ids}-result`} className={sectionTitle}>
              {t.resultTitle}
            </h3>

            {!ready ? (
              <p className="rounded-2xl bg-sidebar-accent p-4 text-sm text-sidebar-foreground/70">
                {hasGlucose ? t.invalid : t.noReading}
              </p>
            ) : isHypo ? (
              <div
                role="alert"
                className="flex flex-col gap-2 rounded-2xl bg-destructive/15 p-4 text-destructive"
              >
                <p className="flex items-center gap-2 font-semibold">
                  <AlertTriangle className="h-5 w-5 shrink-0" aria-hidden="true" />
                  {t.hypoTitle}
                </p>
                <p className="text-sm leading-relaxed">{t.hypoRule}</p>
                <p className="text-sm">
                  {t.carbsNeeded}:{" "}
                  <span className="font-semibold tabular-nums">
                    {carbsNeeded} {t.grams}
                  </span>
                </p>
              </div>
            ) : (
              <div className="flex flex-1 flex-col gap-3">
                <div className="flex flex-1 flex-col justify-between gap-4 rounded-2xl bg-primary p-4 text-primary-foreground">
                  <span className="text-sm font-medium">{t.totalDose}</span>
                  <span className="text-5xl font-semibold tabular-nums leading-none sm:text-6xl">
                    {total.toFixed(1)}
                    <span className="ml-1.5 text-lg font-medium">{t.units}</span>
                  </span>
                </div>

                <dl className="flex flex-col gap-2 rounded-2xl bg-sidebar-accent p-3 text-sm sm:p-4">
                  <ResultRow label={t.carbDose} value={fmtU(carbSteps)} />
                  <ResultRow label={t.correctionDose} value={fmtU(correction)} />
                  {iobApplied > 0 && <ResultRow label={t.iobDeduction} value={fmtU(-iobApplied)} />}
                </dl>

                {carbsG > 0 && carbsG % ratioG > 1e-9 && (
                  <p className="text-xs text-sidebar-foreground/60">
                    {t.carbsRounded.replace("{grams}", String(carbSteps * ratioG))}
                  </p>
                )}

                {total === 0 && (
                  <p className="text-sm text-sidebar-foreground/70">{t.noInsulinNeeded}</p>
                )}
              </div>
            )}
          </section>

          <fieldset className="flex min-w-0 flex-col gap-3">
            <legend className={`${sectionTitle} mb-3 p-0`}>{t.parameters}</legend>
            <NumberField
              id={`${ids}-ratio`}
              label={t.ratio}
              hint={t.ratioHint}
              suffix={`${t.grams} ${t.perUnit}`}
              value={ratio}
              onChange={setRatio}
              step="0.5"
              compactHint
            />
            <NumberField
              id={`${ids}-isf`}
              label={t.sensitivity}
              hint={t.sensitivityHint}
              suffix={`${glucoseUnit} ${t.perUnit}`}
              value={sensitivity}
              onChange={setSensitivity}
              step={glucoseStep}
              compactHint
            />
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
              <NumberField
                id={`${ids}-low`}
                label={t.rangeLow}
                suffix={glucoseUnit}
                value={rangeLow}
                onChange={setRangeLow}
                step={glucoseStep}
              />
              <NumberField
                id={`${ids}-high`}
                label={t.rangeHigh}
                suffix={glucoseUnit}
                value={rangeHigh}
                onChange={setRangeHigh}
                step={glucoseStep}
              />
            </div>
            <NumberField
              id={`${ids}-iob`}
              label={t.iob}
              hint={t.iobHint}
              suffix={t.units}
              value={iob}
              onChange={setIob}
              step="0.5"
              compactHint
            />
            <p className="text-xs text-sidebar-foreground/60">{t.sessionNote}</p>
          </fieldset>
        </div>

        {paramsValid && (
          <DoseTable
            ratio={ratioG}
            params={params}
            lowThreshold={settings.lowThreshold}
            unit={unit}
            activeSteps={ready && !isHypo ? Math.min(steps, MAX_CORRECTION_STEPS) : null}
            activeCarbSteps={ready && !isHypo ? Math.min(carbSteps, MAX_CARB_STEPS) : null}
          />
        )}

        <p className="flex items-start gap-2 text-xs leading-relaxed text-sidebar-foreground/60">
          <Info className="mt-0.5 h-4 w-4 shrink-0" aria-hidden="true" />
          {t.disclaimer}
        </p>
      </div>
    </Modal>
  );
}
